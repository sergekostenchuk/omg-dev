## 7. СЛУЖЕБНЫЙ СКРИПТ — пересоздать в новом scratchpad как `read_channel.py`

```python
# -*- coding: utf-8 -*-
# Показать записи журнала ревью начиная с номера N; вердикты Codex сохранить дословно в ~/REPORTS/review/<имя>.
import io, json, os, sys
start = int(sys.argv[1]); save = sys.argv[2] if len(sys.argv) > 2 else None
ls = io.open(os.path.expanduser('~/.asvp-review/channel.ndjson'), encoding='utf-8').read().splitlines()
print(u'строк в журнале: %d' % len(ls))
for i, l in enumerate(ls[start - 1:], start):
    d = json.loads(l)
    print(u'#%d %s %s→%s %s [%s]' % (i, d.get('ts', '')[5:16], d.get('from'), d.get('to'), d.get('kind'), d.get('topic')))
    if d.get('from') == 'codex-omg':
        if save:
            out = os.path.expanduser('~/REPORTS/review/' + save)
            io.open(out, 'w', encoding='utf-8').write(u'# %s · запись журнала №%d, дословно\n\n' % (d.get('topic'), i) + d['text'])
        print(d['text'])
```
